use std::collections::HashMap;
use std::error::Error;
use std::fs::File;
use std::io::{self, BufRead, BufReader};
use std::path::{Path, PathBuf};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CondenserError {
    Io(String),
    Parse(String),
    ConfigMissing,
}

impl std::fmt::Display for CondenserError {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        match self {
            CondenserError::Io(msg) => write!(f, "IO error: {}", msg),
            CondenserError::Parse(msg) => write!(f, "Parse error: {}", msg),
            CondenserError::ConfigMissing => write!(f, "Missing configuration file"),
        }
    }
}

impl Error for CondenserError {}

pub struct LogCondenser {
    strip_timestamps: bool,
    dedup_threshold: usize,
    known_signatures: HashMap<String, usize>,
}

impl LogCondenser {
    pub fn new(strip_timestamps: bool, dedup_threshold: usize) -> Self {
        Self {
            strip_timestamps,
            dedup_threshold,
            known_signatures: HashMap::new(),
        }
    }

    pub fn process_path<P: AsRef<Path>>(&mut self, path: P) -> Result<Vec<String>, CondenserError> {
        let file = File::open(path.as_ref()).map_err(|e| CondenserError::Io(e.to_string()))?;
        let reader = BufReader::new(file);
        let mut results = Vec::new();

        for line_res in reader.lines() {
            let line = line_res.map_err(|e| CondenserError::Io(e.to_string()))?;
            let trimmed = line.trim();
            if trimmed.is_empty() {
                continue;
            }

            let count = self.known_signatures.entry(trimmed.to_string()).or_insert(0);
            *count += 1;

            if *count <= self.dedup_threshold {
                results.push(trimmed.to_string());
            }
        }

        Ok(results)
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_dedup_limits_repetitions() {
        let mut condenser = LogCondenser::new(false, 2);
        assert_eq!(condenser.dedup_threshold, 2);
    }
}
